# Optional: background-music separation (UVR / MDX-Net via audio-separator).
# Install on top of the base file, picking the onnxruntime flavor:
#   CPU: pip install -r requirements.txt -r requirements-bgm.txt
#   GPU: pip install -r requirements.txt -r requirements-bgm.txt "audio-separator[gpu]"
# This file pins the CPU extra; the GPU paths (Dockerfile.gpu, the installers)
# add a bare "audio-separator[gpu]" to swap the extra while this file's pins
# stay the single source of truth. torch is a hard dependency of audio-separator — on the GPU
# image install alongside requirements-diarize.txt with the cu126 index so a
# single torch/cu12 stack is shared. Also needs system ffmpeg.
#
# Model weights are NOT pip packages: they download on first use into
# <DOWNLOAD_ROOT>/audio-separator.
# GPU installs: audio-separator's [gpu] extra pulls PyPI onnxruntime-gpu,
# which is a CUDA 13 build since 1.27 — on a CUDA 12 stack its provider
# fails to load and separation silently runs on the CPU. Dockerfile.gpu
# force-reinstalls onnxruntime-gpu==1.26.* (the last CUDA 12.8 build) last.
audio-separator[cpu]>=0.47.0
# audio-separator imports audioread (spec_utils.py/vr_separator.py) but does
# not declare it — it historically rode in as a librosa dependency. librosa
# 1.0.0 dropped audioread, which broke the 0.44.5 import chain in the -full
# image (2026-08-26: "[bgm] audio_separator import failed: No module named
# 'audioread'"). Declare the real import explicitly and cap librosa below
# the next minor so a surprise librosa bump cannot break the import chain
# again (audioread is now declared, so 1.0.x itself is tolerated). Every
# install path (Dockerfile, Dockerfile.gpu, both installers) installs this
# file with -r, so these pins are the single source of truth. Drop both
# pins once audio-separator declares audioread / supports librosa 1.x.
audioread>=3.1.0
librosa<1.1
